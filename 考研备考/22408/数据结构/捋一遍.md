**广度优先搜索**

算法导论中的部分：
- 回顾思路：证明该方法得的的路径长度>=最小的；同时<=最小的；得到等于最小的
- 广搜同样适用于有向图
- 下面代码是使用广度优先的思想实现的二叉树的层次遍历
``` cpp
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        //广度优先搜索转化的 ---> 层次遍历
        queue<TreeNode*>q;
        q.push(root);
        vector<vector<int>>res;
        if(root == nullptr) return res;
        while(!q.empty())
        {
            int num = q.size();
            vector<int>cur_row;
            for(int i = 0 ; i<num;i++)
            {
            auto cur = q.front();
            q.pop();
            cur_row.push_back(cur->val);
            if(cur->left!=nullptr){q.push(cur->left);}
            if(cur->right!=nullptr){q.push(cur->right);}
            }
            res.push_back(cur_row);
        }
        return res;
    }
};
```

最小生成树：

算法导论部分：
- 生成树：包含全部顶点，连通的子图
- ==多条边权相同==可能得到不同形状，但是总权值相同的MST
- 维护一个边集，不断往里面放安全边，直到n-1条
- Kruskal按边权选择
- Prim选点，每次选连接已经选边集和未选边集最小的边
- 割是将点划分为两组，边的两个顶点一个在A组一个在B组，则这个边跨越割
- 一个割尊重边集A指的是边集A中==没有跨越割的边==
- 在所有跨越割的边中权值最小的，叫做轻边
- 定理
	- A 是某棵 MST 的子集
	- 割 (X,Y) 尊重 A
	- 边 (u,v) 是横跨割 (X,Y) 的一条轻边
	- 则边是安全边 

K算法的思想：
在当前森林 `G_A=(V,A)` 中，取任意一个连通分量 `C`。
如果 `(u,v)` 是从 `C` 连到其他分量的最小权边，那么 `(u,v)` 是安全边。
因为可以构造割：
```
X = C 中的顶点
Y = V - X
```
这个割不会被 `A` 中的边跨越，因为 `A` 的边都在各自分量内部，所以割尊重 `A`。
而 `(u,v)` 是跨这个割的轻边，所以根据安全边定理，它安全。

1. 初始时，每个顶点自成一个树，A = ∅
2. 按边权从小到大考察边
3. 如果边 (u,v) 连接两个不同的树，就加入 A
4. 合并这两个树
5. 直到 A 有 |V|-1 条边

P算法
为什么 Prim 每次选的边是安全边
每一步看割：
```
(U, V-U)
```
当前已选边 `A` 都在 `U` 内部，不会跨越这个割，所以这个割**尊重 A**。
Prim 选的是所有跨越这个割的边中权值最小的一条，也就是这个割的**轻边**。
根据安全边定理：
```
尊重 A 的割上的轻边，一定是 A 的安全边
```

优先队列实现：
struct edge {
  int v, w;
};

struct node {
  int dis, u;

  bool operator>(const node& a) const { return dis > a.dis; }
};

vector<edge> e[MAXN];
int dis[MAXN], vis[MAXN];
priority_queue<node, vector<node>, greater<node>> q;

void dijkstra(int n, int s) {
  memset(dis, 0x3f, (n + 1) * sizeof(int));
  memset(vis, 0, (n + 1) * sizeof(int));
  dis[s] = 0;
  q.push({0, s});
  while (!q.empty()) {
    int u = q.top().u;
    q.pop();
    if (vis[u]) continue;
    vis[u] = 1;
    for (auto ed : e[u]) {
      int v = ed.v, w = ed.w;
      if (vis[v]) continue;//不改已经访问过的点
      if (dis[v] > dis[u] + w) {
        dis[v] = dis[u] + w;
        q.push({dis[v], v});
      }
    }
  }
}

        return;
    }

    L->next->next = NULL;    // 将已排序部分（即第一个结点）与后续断开，形成独立的有序表

    LNode *pre; // 工作指针，用于在有序表中寻找插入位置的前驱

        pre = L;     // 2. 每次都从头结点开始比较

    int data;           
    struct LNode *next;

  void BubbleSort(LinkList L) {
    // 1. 边界条件判断：若链表为空或只有一个结点，无需排序
    if (L == NULL || L->next == NULL || L->next->next == NULL) {
        return;
    }
    LNode *p;
    LNode *tail = NULL; // tail指向已排好序部分的第一个结点（即未排序部分的边界）
    int swapped;        // 优化标记：判断本轮是否发生交换
    do { 
        swapped = 0;
        p = L->next; // 每次从第一个有效结点开始，直到某一次经过一趟之后发现没有swapped!!!

        // 内层循环：两两比较，直到遇到已排好序的边界 tail
        while (p->next != tail) {
            if (p->data > p->next->data) {
                // 交换数据域（408最推荐的稳健写法）
                int temp = p->data;
                p->data = p->next->data;
                p->next->data = temp;
                
                swapped = 1; // 发生了交换
            }
            p = p->next; // 指针后移
        }
        
        // 核心优化：p 此时到达了本轮比较的最后一个结点，它必然是剩余未排序序列中的最大值
        tail = p; 
        
    } while (swapped); // 如果一整轮都没有发生交换，说明已经有序，退出循环
}

void SelectSort(LinkList L) {
    // 1. 边界条件判断
    if (L == NULL || L->next == NULL || L->next->next == NULL) {
        return;
    }

    LNode *h = L->next;     // h 指向当前未排序链表的首结点
    L->next = NULL;         // 已排序链表初始化为空（仅剩头结点 L）
    LNode *tail = L;        // tail 始终指向已排序链表的尾结点

    // 外层循环：只要还有未排序的结点
    while (h != NULL) {
        LNode *min_node = h;    // 记录当前找到的最小结点，初始化为首结点
        LNode *min_pre = NULL;  // 记录最小结点的前驱

        LNode *curr = h;        // 工作指针，用于遍历未排序链表
        LNode *curr_pre = NULL; // 工作指针的前驱

        // 内层循环：遍历未排序部分，寻找最小值及其前驱
        while (curr != NULL) {
            if (curr->data < min_node->data) {
                min_node = curr;
                min_pre = curr_pre;
            }
            curr_pre = curr;
            curr = curr->next;
        }

        // 将最小结点 min_node 从未排序链表 h 中摘除
        if (min_node == h) {
            h = h->next;        // 最小值恰好是未排序链表的第一个结点
        } else {
            min_pre->next = min_node->next; // 最小值在链表中间或尾部
        }

        // 将摘除出来的 min_node 尾插到已排序链表的末尾
        tail->next = min_node;
        tail = min_node;
        tail->next = NULL;      // 确保已排序链表的尾部指针域为 NULL
    }

		  int mid = (low+high)/2;
		  MergeSort(A,low,mid);    //对左侧进行归并
		  MergeSort(A,mid+1,high); //对右侧进行归并
		  Merge(A,low,mid,high) //当上面两个函数返回的时候，说明左右已经归并完了整体进行
	  }//开始的时候会一直深入，直到每一段的长度为1，然后返回！！！
  }
  在merge内部的话就是要使用一个临时的数组B，先将原数组A中的值复制到数组B中，然后使用三指针法归并回A
  
  void Merge(int A[],int low,int mid,int high)
  {
	  int i,j,k;
	  for(k = low;k<=high;k++)
	  {
	  B[k] = A[k];
	  }
	  for(k=low,i=low,j=mid+1;(i<=mid&&j<=high);k++)
	  {
		  if(B[i]<B[j]){
		  A[k]=B[i];i++;
		  }
		  else{
		  A[k]=B[j];j++;
		  }
	  
	  }
	  while(i<=mid)
		  A[k++] = B[i++];
	  while(j<=high)
		  A[k++] = B[j++];
  }
  
  ```

  void CountSort(int A[],int B[],int n,int k)
	  int i,C[k] = {0};
	  for(int i = 0; i<n;i++){C[A[i]]++;}//计数
	  for(int i = 0; i<n;i++){C[i]=C[i]+C[i-1];}//前缀
	  for(int i=n-1;i>=0;i--)
	  {
		  C[A[i]]--;
		  B[C[A[i]]]=A[i];
	  }

**败者树**（找小的）
叶子节点表示参与比赛的，分支节点填写失败的（也就是大的）
![[Pasted image 20260712095851.png|347]]
通过这种方法将原来的k次比较减少为logk次比较
败者树通过保存“历史败者”，实际上**保存了每个子树的局部最优解**。新元素只需要在向上攀升的过程中，击败每一个遇到的局部最优解，就能名正言顺地成为全局最优解。
叶子节点是虚拟的，数组中只存中间节点
如果是奇数的情况可以补充虚拟节点，并设置为infinity，使其称为绝对输者
