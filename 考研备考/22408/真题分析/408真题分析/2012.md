int get_mid(int arr[],int s,int e){return arr[(s+e)/2];}

int mid_num(int arr1[],int arr2[],int s1,int e1,int s2,int e2)
{
	int num1=get_mid(arr1,s1,e1);
	int num2=get_mid(arr2,s2,e2);
	if(num1==num2)return num1;//这里添加两个数组长度为1的时候取较小的为result
	else if(num1>num2){return mid_num(arr1,arr2,s1,(s1+e1)/2-1,(s2+e2)/2,e2);//这里改成正确的位置
	else{return mid_num(arr1,arr2,(s1+e1)/2,e1,s2,(s2+e2)/2-1);}
}
